# Cell 9: Expose gateway and verify MCP paths
from pyngrok import ngrok
import requests

mcp_gateway_tunnel = ngrok.connect(addr="8200", proto="http", hostname=MCP_GATEWAY_HOSTNAME)
MCP_GATEWAY_URL = mcp_gateway_tunnel.public_url
OPENCONTEXT_MCP_URL = f"{MCP_GATEWAY_URL}/opencontext/mcp"
OPENGOV_MCP_URL = f"{MCP_GATEWAY_URL}/opengov/mcp"
OPENGOV_NYS_MCP_URL = f"{MCP_GATEWAY_URL}/opengov-nys/mcp"
print("MCP_GATEWAY_URL:", MCP_GATEWAY_URL)

headers = {
    "Content-Type": "application/json",
    "Accept": "application/json, text/event-stream",
}
ping_payload = {"jsonrpc": "2.0", "id": "1", "method": "ping", "params": {}}
init_payload = {
    "jsonrpc": "2.0",
    "id": "1",
    "method": "initialize",
    "params": {
        "protocolVersion": "2025-03-26",
        "clientInfo": {"name": "colab-check", "version": "0.1.0"},
        "capabilities": {"tools": {}},
    },
}
print("opencontext:", requests.post(OPENCONTEXT_MCP_URL, json=ping_payload, headers=headers, timeout=20).status_code)
print("opengov:", requests.post(OPENGOV_MCP_URL, json=init_payload, headers=headers, timeout=20).status_code)
print("opengov-nys:", requests.post(OPENGOV_NYS_MCP_URL, json=init_payload, headers=headers, timeout=20).status_code)
