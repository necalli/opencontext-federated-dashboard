def sh(cmd, check=True):
    p = subprocess.run(["bash", "-lc", cmd], text=True)
    if check and p.returncode != 0:
        raise RuntimeError(f"Command failed ({p.returncode}): {cmd}")

def sh_out(cmd):
    return subprocess.check_output(["bash", "-lc", cmd], text=True).strip()

def start_bg(name, cmd_list, log_path, cwd=None, env=None):
    logf = open(log_path, "w", encoding="utf-8")
    p = subprocess.Popen(cmd_list, cwd=cwd, env=env, stdout=logf, stderr=subprocess.STDOUT)
    return p, logf

def wait_http(url, timeout=180, headers=None):
    end = time.time() + timeout
    last = None
    while time.time() < end:
        try:
            r = requests.get(url, timeout=8, headers=headers or {})
            if r.status_code < 500:
                return r
            last = f"status={r.status_code}"
        except Exception as e:
            last = str(e)
        time.sleep(2)
    raise RuntimeError(f"Timeout waiting for {url}. Last error: {last}")

def wait_post(url, payload, timeout=180, headers=None):
    end = time.time() + timeout
    last = None
    while time.time() < end:
        try:
            r = requests.post(url, json=payload, timeout=12, headers=headers or {})
            if r.status_code < 500:
                return r
            last = f"status={r.status_code}"
        except Exception as e:
            last = str(e)
        time.sleep(2)
    raise RuntimeError(f"Timeout waiting for POST {url}. Last error: {last}")

def wait_ngrok_https(timeout=120):
    end = time.time() + timeout
    last = None
    while time.time() < end:
        try:
            r = requests.get("http://127.0.0.1:4040/api/tunnels", timeout=5)
            data = r.json()
            for t in data.get("tunnels", []):
                u = str(t.get("public_url", "")).strip()
                if u.startswith("https://"):
                    return u.rstrip("/")
            last = "no https tunnel yet"
        except Exception as e:
            last = str(e)
        time.sleep(2)
    raise RuntimeError(f"Timed out reading ngrok tunnel URL. Last error: {last}")

def safe_json_get(url, timeout=30):
    r = requests.get(url, timeout=timeout, headers={"ngrok-skip-browser-warning": "true"})
    try:
        return r.json()
    except Exception:
        preview = (r.text or "")[:300]
        raise RuntimeError(f"Non-JSON response from {url} | status={r.status_code} | body={preview}")
